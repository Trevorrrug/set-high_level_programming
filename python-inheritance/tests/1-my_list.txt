# 1. My List

## Test print_sorted with sorted list

>>> MyList = __import__('1-my_list').MyList
>>> my_list = MyList([1, 2, 3])
>>> my_list.print_sorted()
[1, 2, 3]

## Test print_sorted with unsorted list

>>> my_list = MyList([4, 2, 3, 1])
>>> my_list.print_sorted()
[1, 2, 3, 4]

## Test that original list is unchanged

>>> my_list
[4, 2, 3, 1]

## Test negative numbers

>>> my_list = MyList([4, -2, 3, -1])
>>> my_list.print_sorted()
[-2, -1, 3, 4]

## Test empty list

>>> my_list = MyList()
>>> my_list.print_sorted()
[]
